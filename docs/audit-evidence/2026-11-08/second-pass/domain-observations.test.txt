import { expect, it, vi } from 'vitest'
const mocks = vi.hoisted(() => ({ upsert: vi.fn(async () => ({ ok: true })), read: vi.fn(async () => ({ ok: true, data: [] })), remove: vi.fn(async () => ({ ok: true })) }))
vi.mock('./data/repositories/loansRepo', () => ({ fetchLoanInstallmentsByLoan: mocks.read, upsertLoanInstallments: mocks.upsert, deleteLoanInstallmentsByIds: mocks.remove }))
vi.mock('./data/repositories/cardsRepo', () => ({ fetchCardsByType: vi.fn() }))
import { syncLoanInstallmentPlan, validateLoanForm } from './pages/LoansPage.helpers'
import { summarizeGoldType } from './utils/goldLedger'

it('ikinci tarama: kabul edilen 360 aylık kredi sessizce 240 taksite kesiliyor', async () => {
  const loan: any = { id: 'l', user_id: 'u', start_date: '2026-01-20', end_date: '2055-12-20', installment_day: 20, monthly_payment: 1000, total_amount: 360000 }
  const form = new FormData()
  for (const [key, value] of Object.entries(loan)) form.set(key, String(value))
  expect(validateLoanForm(form)).toEqual({})
  await syncLoanInstallmentPlan(loan)
  const schedule = (mocks.upsert.mock.calls as any)[0][0]
  expect(schedule).toHaveLength(240)
  expect(schedule.at(-1).due_date).toBe('2045-12-20')
  console.log('AUDIT_LOAN_CAP', { intended: 360, actual: schedule.length, lastDate: schedule.at(-1).due_date })
})

it('ikinci tarama: tarihsel negatif altın havuzu satışın açığını gizliyor', () => {
  const lots: any = [
    { id: 'sell', created_at: '2026-10-01', purchase_date: '2026-10-01', direction: 'sell', gold_type: 'gram', quantity: 4, unit_price: 5000 },
    { id: 'buy', created_at: '2026-10-05', purchase_date: '2026-10-05', direction: 'buy', gold_type: 'gram', quantity: 10, unit_price: 5000 },
  ]
  const summary = summarizeGoldType(lots, 'gram')
  expect(summary.totalQuantity).toBe(10)
  console.log('AUDIT_GOLD_TIMELINE', { buys: 10, sells: 4, reported: summary.totalQuantity })
})
