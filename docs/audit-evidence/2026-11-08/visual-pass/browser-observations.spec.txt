async function mockApp(page: Page, theme: string, overrides: typeof tables = {}) {
  const rows = { ...tables, ...overrides }
  const snapshot = Object.fromEntries(snapshotKeys.map(key => [key, rows[key] ?? []]))
  await page.clock.install({ time: new Date(now) })
  const user = { id: userId, email: 'synthetic@example.test', aud: 'authenticated', role: 'authenticated', created_at: now, app_metadata: {}, user_metadata: {} }
  const token = `${Buffer.from('{"alg":"HS256","typ":"JWT"}').toString('base64url')}.${Buffer.from(JSON.stringify({ sub: userId, role: 'authenticated', aud: 'authenticated', exp: 2100000000 })).toString('base64url')}.synthetic`
  await page.addInitScript(({ user, token, theme, now }) => {
    localStorage.setItem('sb-example-auth-token', JSON.stringify({ access_token: token, refresh_token: 'synthetic', expires_at: 2100000000, expires_in: 3600, token_type: 'bearer', user }))
    localStorage.setItem('theme', theme)
    localStorage.setItem('denge:theme-preference', theme)
    localStorage.setItem('denge:net-worth-snapshot-day', `${user.id}:2026-10-26`)
    localStorage.setItem('fp.marketRates.v1', JSON.stringify({ rates: { USD: { buying: 40, selling: 41 }, GRA: { buying: 5000, selling: 5100 }, CEYREKALTIN: { buying: 8000, selling: 8100 } }, asOf: now, fetchedAt: now }))
  }, { user, token, theme, now })
  await page.route('**/*', route => {
    const url = new URL(route.request().url())
    if (['127.0.0.1', 'localhost'].includes(url.hostname)) {
      if (url.pathname.startsWith('/_vercel/')) return route.fulfill({ body: '', contentType: 'application/javascript' })
      return route.continue()
    }
    if (url.hostname === 'example.supabase.co') {
      const name = url.pathname.split('/').at(-1)!
      let data: unknown = name === 'fetch_finance_snapshot' ? snapshot : name === 'bist-quote' ? { prices: {}, asOf: now } : url.pathname.includes('/rpc/') ? 0 : name === 'user' ? user : rows[name] ?? []
      if (['card_ledger', 'account_ledger', 'card_expenses', 'card_installments', 'card_statement_archives'].includes(name)) {
        const before = url.searchParams.get('id')?.replace(/^lt\./, '')
        const cardId = url.searchParams.get('card_id')?.replace(/^eq\./, '')
        data = (rows[name] ?? []).filter(row => (!before || String(row.id) < before) && (!cardId || row.card_id === cardId))
          .toSorted((a, b) => String(b.id).localeCompare(String(a.id))).slice(0, Number(url.searchParams.get('limit') ?? 1000))
      }
      return route.fulfill({ json: data })
    }
    return route.abort()
  })
}

for (const theme of ['light', 'dark']) test(`visual contrast and targets ${theme}`, async ({ page }) => {
  await page.setViewportSize({ width: 390, height: 900 })
  await mockApp(page, theme)
  await page.goto('/analiz#ay-kapanisi')
  await expect(page.getByText('Net değer trendi', { exact: true })).toBeVisible()
  const contrast = await page.locator('main').evaluate(main => {
    const c = document.createElement('canvas').getContext('2d')!
    function rgba(value: string): number[] { c.clearRect(0,0,1,1); c.fillStyle = value; c.fillRect(0,0,1,1); const a = c.getImageData(0,0,1,1).data; return [a[0],a[1],a[2],a[3]/255] }
    function over(a: number[], b: number[]) { return [0,1,2].map(i => a[i]*a[3]+b[i]*(1-a[3])).concat(1) }
    function lum(a: number[]) { return a.slice(0,3).map(v=>v/255).map(v=>v<=0.04045?v/12.92:((v+0.055)/1.055)**2.4).reduce((s,v,i)=>s+v*[0.2126,0.7152,0.0722][i],0) }
    const results: unknown[] = []
    for (const e of main.querySelectorAll<HTMLElement>('p,span,button,a,li,h2,h3,label')) {
      if (e.childElementCount || !e.textContent?.trim() || e.closest('[disabled],[aria-hidden="true"],.sr-only') || !e.getBoundingClientRect().height || e.closest('details:not([open])') && !e.closest('summary')) continue
      const chain: Element[] = []; for(let n: Element|null=e;n;n=n.parentElement) chain.unshift(n)
      let bg=[255,255,255,1],opacity=1
      for(const n of chain) { const s=getComputedStyle(n); bg=over(rgba(s.backgroundColor),bg); opacity*=Number(s.opacity) }
      const style=getComputedStyle(e), fg=rgba(style.color); fg[3]*=opacity
      const rendered=over(fg,bg), l1=lum(rendered), l2=lum(bg), ratio=(Math.max(l1,l2)+0.05)/(Math.min(l1,l2)+0.05)
      const fs=parseFloat(style.fontSize), large=fs>=24 || fs>=18.66 && Number(style.fontWeight)>=700
      if(ratio<(large?3:4.5))results.push({text:e.textContent,ratio,fg:style.color,bg,opacity,font:style.fontSize,class:e.className})
    }
    return results
  })
  const targets = await page.locator('main button,main a').evaluateAll(els=>els.filter(e=>e.getBoundingClientRect().height>0).map(e=>{const r=e.getBoundingClientRect(),s=getComputedStyle(e,'::before'),a=getComputedStyle(e,'::after');return {text:e.textContent?.trim(),aria:e.getAttribute('aria-label'),width:r.width,height:r.height,before:s.content,after:a.content}}).filter(x=>x.height<44&&x.before==='none'&&x.after==='none'))
  await evidence(page, `probe-contrast-${theme}`, {contrast,targets})
})

  await mockApp(page,'light',mode==='empty'?Object.fromEntries(Object.keys(tables).map(k=>[k,[]])):{})
